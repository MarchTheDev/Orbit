```rust
/// Find the Amazon Games installs Nile recorded.
///
/// Amazon Luna streams its games, so there is nothing of its own on disk. What
/// is on disk is what the Amazon Games client downloaded, and the only readable
/// record of that is Nile's: the unofficial client keeps two files side by side,
/// `library.json` for what an account owns and `installed.json` for what is
/// installed where. The official Amazon Games app keeps no such file.
///
/// Neither list carries both halves - the entitlements have titles and the
/// installs have paths - so they are joined on the product id.
fn scan_amazon(games: &mut Vec<LauncherGame>) {
    let mut found: Vec<LauncherGame> = Vec::new();
    for root in config_roots("nile") {
        push_amazon_records(
            &root.join("library.json"),
            &root.join("installed.json"),
            &mut found,
        );
    }
    games.extend(dedupe_by_install(found));
}
```

```rust
fn push_amazon_records(library: &Path, installed: &Path, games: &mut Vec<LauncherGame>) {
    let titles = amazon_titles(library);
    let Ok(text) = std::fs::read_to_string(installed) else { return; };
    let Ok(json) = serde_json::from_str::<Value>(&text) else { return; };
    let Value::Array(entries) = json else { return; };
    for entry in &entries {
        let Some(id) = entry.get("id").and_then(Value::as_str) else { continue; };
        let Some(path) = entry.get("path").and_then(Value::as_str).filter(|p| !p.is_empty()) else { continue; };

        games.push(LauncherGame { launcher: "Amazon Luna".into(), name, install_dir: path.to_string(), exe_path: None, size_bytes: 0 });
    }
}
```

```rust
/// The titles Nile cached, by product id.
fn nile_titles(library: &Path) -> BTreeMap<String, String> {
    let Ok(text) = std::fs::read_to_string(library) else {
        return BTreeMap::new();
    };
    let Ok(list) = serde_json::from_str::<Vec<serde_json::Value>>(&text) else {
        return BTreeMap::new();
    };
    list.iter()
        .filter_map(|entry| {
            let product = entry.get("product")?;
            let id = product.get("id").and_then(|value| value.as_str())?;
            let title = product.get("title").and_then(|value| value.as_str()).unwrap_or("");
            (!id.is_empty()).then(|| (id.to_string(), title.trim().to_string()))
        })
        .collect()
}

/// Add every Nile install whose folder is really there.
fn push_amazon_records(library: &Path, installed: &Path, games: &mut Vec<LauncherGame>) {
    let titles = nile_titles(library);

        return;

        return;

            continue;
        };

            .get("path")
            .and_then(|value| value.as_str())
```
